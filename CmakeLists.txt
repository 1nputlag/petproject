cmake_minimum_required(VERSION 3.10)

project(AimTrainer VERSION 1.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

#шукаємо пакети SDL2 на комп'ютері
find_package(SDL2 REQUIRED)
find_package(SDL2_ttf REQUIRED)

#вказуємо, що наш виконуваний файл AimTrainer збирається з цих .cpp файлів
add_executable(AimTrainer main.cpp)

#прив'язуємо знайдені бібліотеки SDL2 до нашого проєкту
target_link_libraries(AimTrainer PRIVATE SDL2::SDL2 SDL2_ttf::SDL2_ttf)
